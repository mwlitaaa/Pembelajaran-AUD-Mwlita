function ucap(teks, rate=0.92){

  if(!suaraAktif) return;

  if(!('speechSynthesis' in window)) return;

  try{

    window.speechSynthesis.cancel();

    const u = new SpeechSynthesisUtterance(teks);

    u.lang = 'id-ID';

    u.rate = rate;

    u.pitch = 1.15;

    u.volume = 1;

    const v = window.speechSynthesis.getVoices().find(v=>/id|Indo/i.test(v.lang));

    if(v) u.voice = v;

    window.speechSynthesis.speak(u);

  }catch(e){}

}

let idx = 0;          // index soal utama

let skor = 0;

let subStep = null;   // untuk soal lanjutan (perasaan)

let terkunci = false; // mencegah klik ganda

/* ---------- Render halaman ---------- */

function showPage(id){

  document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));

  document.getElementById(id).classList.add('active');

}

/* ---------- Mulai ---------- */

function startGame(){

  idx = 0; skor = 0; subStep = null;

  updateSkor();

  showPage('page-quiz');

  renderSoal();

  ucap('Ayo mulai petualangan! Bantu Lala memilih tindakan yang baik.');

}

/* ---------- Render satu soal ---------- */

function renderSoal(){

  const s = SOAL[idx];

  const data = subStep ? subStep : s;

  document.getElementById('pill-step').textContent = s.label;

  document.getElementById('q-situ').textContent = data.situasi;

  document.getElementById('q-ask').textContent = data.tanya;

  document.getElementById('q-art').innerHTML = ART[data.art] ? ART[data.art]() : '';

  // progress

  const total = SOAL.length;

  const progresSekarang = subStep ? idx + 0.5 : idx;

  document.getElementById('prog').style.width = ((progresSekarang/total)*100).toFixed(1) + '%';

  // opsi

  const wrap = document.getElementById('q-options');

  wrap.innerHTML = '';

  terkunci = false;

  const huruf = ['A','B','C'];

  data.opsi.forEach((o,i)=>{

    const b = document.createElement('button');

    b.className = 'opt';

    b.innerHTML = `<span class="letter">${huruf[i]}</span><span>${o.teks}</span>`;

    b.dataset.index = i;

    b.onclick = ()=>pilihJawaban(b, o, data);

    wrap.appendChild(b);

  });

  // reset feedback

  const fb = document.getElementById('q-feedback');

  fb.className = 'feedback';

  fb.querySelector('#fb-text').textContent = '';

  document.getElementById('next-btn').classList.remove('show');

  document.getElementById('next-btn').textContent = 'LANJUT ➜';

  // suara otomatis

  setTimeout(()=>ucap(data.suaraSituasi), 320);

}

/* ---------- Saat anak memilih jawaban ---------- */

function pilihJawaban(btn, opsi, data){

  if(terkunci) return;

  terkunci = true;

  const semua = document.querySelectorAll('#q-options .opt');

  semua.forEach(b=>b.disabled = true);

  const fb = document.getElementById('q-feedback');

  const fbText = document.getElementById('fb-text');

  const fbEmo = document.getElementById('fb-emo');

  if(opsi.benar){

    btn.classList.add('correct');

    semua.forEach(b=>{ if(b!==btn) b.classList.add('dim'); });

    skor += SKOR_PER_BENAR;

    updateSkor();

    fb.className = 'feedback ok show';

    fbEmo.textContent = opsi.emo || '🎉';

    fbText.textContent = opsi.fb || 'Hebat! Jawabanmu tepat!';

    ucap(opsi.fbSuara || opsi.fb || 'Hebat! Jawabanmu tepat!');

    confetti();

    popBintang();

    const nextBtn = document.getElementById('next-btn');

    if(opsi.lanjut){

      // ada pertanyaan lanjutan

      nextBtn.classList.add('show');

      nextBtn.textContent = 'LANJUT ➜';

      nextBtn.onclick = ()=>{

        subStep = opsi.lanjut;

        renderSoal();

        setTimeout(()=>ucap(opsi.lanjut.suaraSituasi), 320);

      };

    } else if(subStep){

      // selesai sub-step

      nextBtn.classList.add('show');

      nextBtn.textContent = 'LANJUT ➜';

      nextBtn.onclick = ()=>{ subStep = null; idx++; nextStep(); };

    } else {

      nextBtn.classList.add('show');

      nextBtn.textContent = 'LANJUT ➜';

      nextBtn.onclick = ()=>{ idx++; nextStep(); };

    }

  } else {

    btn.classList.add('wrong');

    fb.className = 'feedback no show';

    fbEmo.textContent = '🤔';

    fbText.textContent = 'Belum tepat. Coba pilih jawaban yang lain ya!';

    ucap('Belum tepat. Coba pilih jawaban yang lain ya!');

    // beri kesempatan mencoba lagi

    setTimeout(()=>{

      btn.disabled = true;

      btn.classList.add('dim');

      const sisa = [...document.querySelectorAll('#q-options .opt')].filter(b=>!b.classList.contains('dim') && b!==btn);

      sisa.forEach(b=>{ b.disabled = false; });

      terkunci = false;

      fb.className = 'feedback';

    }, 1200);

  }

}

/* ---------- Lanjut ---------- */

function nextStep(){

  if(idx >= SOAL.length){

    tampilkanSelesai();

  } else {

    renderSoal();

  }

}

/* ---------- Skor ---------- */

function updateSkor(){

  document.getElementById('pill-score').textContent = '⭐ ' + skor + ' poin';

}

/* ---------- Efek ---------- */

function confetti(){

  const fx = document.getElementById('fx');

  const emojis = ['⭐','🌟','✨','🎉','💖','🎈','🍭','🌈'];

  for(let i=0;i<26;i++){

    const el = document.createElement('div');

    el.className = 'conf';

    el.textContent = emojis[Math.floor(Math.random()*emojis.length)];

    el.style.left = Math.random()*100 + 'vw';

    el.style.fontSize = (18 + Math.random()*22) + 'px';

    el.style.animationDuration = (1.6 + Math.random()*1.6) + 's';

    el.style.animationDelay = (Math.random()*0.35) + 's';

    fx.appendChild(el);

    setTimeout(()=>el.remove(), 3600);

  }

}

function popBintang(){

  const el = document.createElement('div');

  el.className = 'pop-star';

  el.textContent = '🌟';

  document.body.appendChild(el);

  setTimeout(()=>el.remove(), 1100);

}

/* ---------- Selesai ---------- */

function tampilkanSelesai(){

  showPage('page-end');

  document.getElementById('end-score').textContent = skor;

  document.getElementById('end-max').textContent = SOAL.length * SKOR_PER_BENAR;

  document.getElementById('end-lala').innerHTML = ART.end();

  const maks = SOAL.length * SKOR_PER_BENAR;

  const rasio = skor / maks;

  let bintang = 1;

  if(rasio >= 0.85) bintang = 3;

  else if(rasio >= 0.6) bintang = 2;

  document.getElementById('end-stars').textContent = '⭐'.repeat(bintang) + '☆'.repeat(3-bintang);

  setTimeout(()=>{

    ucap('Selamat! Kamu sudah membantu Lala melewati berbagai kegiatan sehari-hari. ' +

         'Menjadi anak yang baik berarti belajar peduli, berbagi, bekerja sama, dan menghargai orang lain.');

  }, 400);

  // confetti perayaan

  for(let i=0;i<3;i++){

    setTimeout(confetti, i*400);

  }

}
